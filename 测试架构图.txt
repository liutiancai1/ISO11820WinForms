┌──────┬──────┬──────────────┬──────┬──────────┬─────────────────────────┐
│ 地址 │ 地址 │   功能名称   │ 读写 │ 数据类型 │         说明            │
│ (Dec)│ (Hex)│              │      │          │                         │
├──────┼──────┼──────────────┼──────┼──────────┼─────────────────────────┤
│  0   │ 0x00 │ 目标温度(SV) │  写  │  UInt16  │ 温度×10, 如750°C=7500   │
│  2   │ 0x02 │ 手动输出(MV) │  写  │  UInt16  │ 0~25600 = 0%~100%       │
│  56  │ 0x38 │ 控制模式     │  写  │  UInt16  │ 3=手动, 4=PID自动       │
│ 257  │ 0x101│ PID输出值    │  读  │  UInt16  │ 0~25600 = 0%~100%       │
│ 258  │ 0x102│ 当前温度(PV) │  读  │  UInt16  │ 温度×10, 如745.2°C=7452 │

┌─────────────────────────────────────────────────────────────────────────────┐
│                           程序状态机流程                                     │
└─────────────────────────────────────────────────────────────────────────────┘

    ┌─────────┐
    │  Idle   │  空闲状态（程序启动后）
    └────┬────┘
         │
         │ 用户点击"开始加热"
         │
         ▼
    ┌─────────┐
    │Preparing│  准备中（升温至750°C并稳定）
    └────┬────┘
         │
         │ 条件满足：
         │ • 温度在 745~755°C 持续 10 分钟
         │ • 温度漂移 < 2°C/10min
         │ • 温度偏差 < 1°C
         │
         ▼
    ┌─────────┐
    │  Ready  │  就绪状态（可以开始试验）
    └────┬────┘
         │
         │ 用户点击"开始试验"
         │
         ▼
    ┌─────────┐
    │Recording│  记录中（采集试验数据）
    └────┬────┘
         │
         │ 条件满足：
         │ • 温度漂移 < 2°C/10min
         │ • 或达到最大试验时间
         │
         ▼
    ┌─────────┐
    │Complete │  完成（保存数据，生成报告）
    └─────────┘

             • 恒功率输出

Core/ApparatusManipulator.cs
    ├─ EstablishConnection()      // 连接 PID 控制器
    ├─ SendPidModuleCmd()         // 发送 Modbus 命令
    ├─ StartHeating()             // 开始加热
    ├─ StopHeating()              // 停止加热
    ├─ SwitchToPID()              // 切换为 PID 模式
    ├─ SwitchToManual()           // 切换为手动模式
    ├─ GetCurrentTemp()           // 读取当前温度
    └─ GetPidOutput()             // 读取 PID 输出

Core/TestMaster.cs
    ├─ DoWork()                   // 状态机主循环
    ├─ CheckStartCriteria()       // 检查试验开始条件
    ├─ CheckTerminateCriteria()   // 检查试验终止条件
    ├─ StartRecording()           // 开始记录数据
    └─ PostTestProcess()          // 试验后处理

Services/SensorSimulator.cs
    ├─ Update()                   // 更新模拟数据
    ├─ StartHeating()             // 开始加热模拟
    └─ UpdateHeatingPhase()       // 更新加热阶段

appsettings.json
    ├─ Hardware.PidPort           // PID 控制器端口
    ├─ Simulation.EnableSimulation // 启用仿真模式
    └─ Simulation.SimulatePidController // 仿真 PID 控制器
